/// Forward projection. lon/lat/lambda0 in radians. Returns (x, y):
/// x = 1.5*(lon-lambda0 normalized to (-pi,pi]) * sqrt(1/3 - (lat/pi)^2), y = lat.
pub fn forward(lon: f64, lat: f64, lambda0: f64) -> (f64, f64);
/// Inverse: returns (lon-lambda0 in (-pi,pi], lat). Total (sqrt arg never zero).
pub fn inverse(x: f64, y: f64) -> (f64, f64);
pub fn normalize_lon(lon: f64) -> f64; // any radians -> (-pi, pi]

/// Layout of the map oval over a dot grid (dots are ~square: cells are 1:2, braille 2x4).
pub struct MapFrame {
    pub dots_w: usize, pub dots_h: usize,
    pub lambda0: f64, // radians
    pub x_min: f64, pub x_max: f64, pub y_min: f64, pub y_max: f64,
}
impl MapFrame {
    /// Uniform scale; whole oval (|x| <= sqrt(3)/2*sqrt(1/3-(y/pi)^2)*... i.e. bounding box
    /// x in [-sqrt(3)*PI/4... use exact oval bbox: x_max = sqrt(3)/2 * sqrt(1/3 - 0)*? 
    /// Simply: bbox x in [-(sqrt(3)*PI)/2 ... wait: at phi=0 width half = (3/2)*(pi)*sqrt(1/3) = sqrt(3)*pi/2.
    /// bbox: x in [-sqrt(3)*PI/2 * (1/2)? no: x(phi)=1.5*lon*sqrt(1/3-(phi/pi)^2); max |x| over lon=pi, phi=0
    ///   => sqrt(3)*pi/2 at equator; at pole half of that. Bbox: x in [-sqrt(3)*pi/2, +sqrt(3)*pi/2], y in [-pi/2, pi/2].
    pub fn new(dots_w: usize, dots_h: usize, lambda0_deg: f64) -> Self;

    pub fn dot_to_lonlat(&self, dx: usize, dy: usize) -> Option<(f64, f64)>;
    /// Nearest dot for (lon_deg, lat_deg) degrees in. None if outside the oval.
    pub fn lonlat_to_dot(&self, lon_deg: f64, lat_deg: f64) -> Option<(usize, usize)>;

## coast.rs + coast_data.rs — embedded land polygons

```rust
// coast.rs
pub type Ring = Vec<(f64, f64)>; // (lat_deg, lon_deg) closed rings, even-odd semantics
pub fn decode(data: &str) -> Vec<Ring>; // base85 -> zigzag varints (centidegrees) -> absolute 2-decimal degrees
// coast_data.rs (generated by examples/datagen.rs between markers):
pub static LAND_DATA: &str = "...";
```

Encoding (datagen): each ring: varint vertex count, then per vertex zigzag varint deltas of round(deg*100). Varint: 7 bits/byte, continuation high bit, little-endian groups. Base85: Z85-like custom alphabet, pad with 'z'-sentinel handled by leading count of total rings. Keep it simple & documented in coast.rs.

}

Unicode braille bit order: dot1=0x01 dot2=0x02 dot3=0x04 dot4=0x08 dot5=0x10 dot6=0x20 dot7=0x40 dot8=0x80.

## frame.rs — frame composition + writers

```rust
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum Style { Blank, DaySea, DayLand, TwiSea, TwiLand, NightSea, NightLand, Terminator, Clock }
#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub struct Cell { pub sym: char, pub style: Style }
impl Cell { pub const BLANK: Cell; }
#[derive(Clone, PartialEq)]
pub struct Frame { pub w: usize, pub h: usize, pub cells: Vec<Cell> } // row-major
impl Frame { pub fn new(w: usize, h: usize, fill: Cell) -> Self; pub fn get(&self, x: usize, y: usize) -> Cell; pub fn set(&mut self, x: usize, y: usize, c: Cell); }

pub struct RenderParams<'a> {
    pub w: usize, pub h: usize,            // terminal size in cells
    pub lambda0_deg: f64,
    pub rings: &'a [coast::Ring],
    pub sun: solar::Sun,
    pub draw_twilight_curve: bool,
    pub ascii: bool,                        // '#' land, '.' terminator instead of braille
    pub clock_line: String,                 // pre-rendered, may be truncated by composer
}
pub fn compose(p: &RenderParams) -> Frame;   // 1-cell margin, bottom clock line, map between
pub fn clock_line(now: chrono::DateTime<chrono::FixedOffset>, sun: &solar::Sun) -> String;
// format: "Wed 2026-09-26 14:03:22 UTC+02:00 · ☉ 12.3°S 45.6°W" (☉ position in deg N/S E/W)
/// Diff writer: ANSI bytes updating changed runs only; None prev = full frame. color=false => plain text rows (used by plain path too).
pub fn render_ansi(prev: Option<&Frame>, next: &Frame, color: bool) -> Vec<u8>;
/// One-shot plain output: full frame rows + trailing newline. color only if enabled.
pub fn render_plain(f: &Frame, color: bool) -> Vec<u8>;
```

Style→SGR: DayLand bright/bold, Night* dim, Twilight intermediate, Terminator accent (e.g. bright yellow/red), indexed 256-colors: pick tasteful constants. With color=false: dim attribute only.
